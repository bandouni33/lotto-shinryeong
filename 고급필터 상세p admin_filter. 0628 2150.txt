# ⚙️ 3. 완벽 정렬 3단 대시보드 로직 (가로줄 단위 렌더링)
# ==========================================================
def draw_premium_pattern(title, options, icon="🔹"):
    with st.container(border=True):
        st.markdown(f'<div class="premium-title"><span>{icon}</span> <span>{title}</span></div>', unsafe_allow_html=True)
        cols = st.columns(len(options))
        for i, opt in enumerate(options):
            cols[i].checkbox(opt, key=f"{title}_{opt}")
